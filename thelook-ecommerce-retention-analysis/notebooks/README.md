# SQL Queries & Business Questions

This file documents all key SQL queries used in the project, each tied to a specific business question. It helps demonstrate how real-world questions were translated into data-driven answers using SQL and cohort analysis.

---

## Business Question
 
**1. How many unique users, orders, and total sales occurred per order status each month?**

```sql
-- Calculate metrics for each month and status
SELECT
  DATE_TRUNC(CAST(created_at AS DATE), MONTH) AS month_year,  -- Extract month-year
  status,                                                     -- Order status (e.g., Complete, Cancelled, Returned)
  COUNT(DISTINCT user_id) AS total_unique_users,              -- Number of unique users
  COUNT(DISTINCT order_id) AS total_orders,                   -- Number of orders
  SUM(sale_price) AS total_sale_price                         -- Total sale price
FROM `bigquery-public-data.thelook_ecommerce.order_items` AS oi
WHERE DATE_TRUNC(created_at, MONTH) BETWEEN '2019-01-01' AND '2022-08-01'
GROUP BY 1, 2
ORDER BY 1, 2;
```

**2. Who are the users that returned orders in August 2022?**

```sql
-- Retrieve user information for users who made a return in August 2022
SELECT
  u.id AS id,                  -- User ID
  u.email AS email,            -- User's email address
  u.first_name AS first_name,  -- User's first name
  u.last_name AS last_name     -- User's last name
FROM `bigquery-public-data.thelook_ecommerce.users` u
INNER JOIN `bigquery-public-data.thelook_ecommerce.orders` o
  ON u.id = o.user_id
WHERE DATE_TRUNC(o.created_at, MONTH) BETWEEN '2022-08-01' AND '2022-08-31'
  AND o.status = 'Returned';
```

**3. What are the top 5 most and least profitable products?**

```sql
with product_summary as (
  -- subquery to calculate the total revenue, number of products bought, and product cost for each product
  select
    oi.product_id,
    sum(oi.sale_price) as total_revenue, -- Total revenue generated by the product
    count(oi.product_id) as product_bought, -- Total number of times the product has been bought
    p.cost -- Cost of the product
  from
    `bigquery-public-data.thelook_ecommerce.order_items` as oi -- alias 'oi' for order_items table
  inner join
    `bigquery-public-data.thelook_ecommerce.products` as p -- alias 'p' for products table
  on
    oi.product_id = p.id
  where
    oi.status in ('Shipped', 'Processing', 'Complete')
  group by
    oi.product_id,
    p.cost
),

product_profit as (
  -- subquery to calculate the profit for each product based on the total revenue and cost
  select
    product_id,
    total_revenue - (product_bought * cost) as profit -- Profit generated by the product (total revenue - total cost)
  from
    product_summary
),

-- retrieve the top 5 most profitable products
top_most_profitable_products as (
  select
    pd.id,
    pd.name, -- Product name
    pd.retail_price, -- Retail price of the product
    pd.cost, -- Cost of the product
    pp.profit, -- Profit generated by the product
    rank() over (order by pp.profit desc) as rank, -- Rank of the product based on profit (descending)
    'most profitable' as least_or_most -- Indicator column to identify 'most profitable' products
  from
    `bigquery-public-data.thelook_ecommerce.products` as pd -- alias 'pd' for products table in main query
  inner join
    product_profit as pp -- alias 'pp' for product_profit CTE
  on
    pd.id = pp.product_id
  order by
    pp.profit desc
  limit
    5
),

-- retrieve the top 5 least profitable products
top_least_profitable_products as (
  select
    pd.id,
    pd.name, -- Product name
    pd.retail_price, -- Retail price of the product
    pd.cost, -- Cost of the product
    pp.profit, -- Profit generated by the product
    rank() over (order by pp.profit asc) as rank, -- Rank of the product based on profit (ascending)
    'least profitable' as least_or_most -- Indicator column to identify 'least profitable' products
  from
    `bigquery-public-data.thelook_ecommerce.products` as pd -- alias 'pd' for products table in main query
  inner join
    product_profit as pp -- alias 'pp' for product_profit CTE
  on
    pd.id = pp.product_id
  order by
    pp.profit
  limit
    5
)

-- combine the results of both top most profitable and top least profitable products
select * from top_most_profitable_products
union all
select * from top_least_profitable_products;
```
